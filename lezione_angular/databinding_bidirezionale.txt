import { Component } from '@angular/core';
import { FormsModule } from '@angular/forms'; // <-- OBBLIGATORIO per usare ngModel

export interface Utente {
nome: string;
cognome: string;
}

@Component({
selector: 'app-root',
standalone: true,
imports: [FormsModule], // <-- Lo registriamo qui
templateUrl: './app.component.html',
styleUrls: ['./app.component.css']
})

<button
(click)="inviaDati()"
style="flex: 2; background-color: #22c55e; color: white; border: none; padding: 10px 15px; border-radius: 4px; cursor: pointer; font-weight: bold;"
>
Invia Form
</button>

<button
(click)="pulisciCampi()"
style="flex: 1; background-color: #ef4444; color: white; border: none; padding: 10px 15px; border-radius: 4px; cursor: pointer; font-weight: bold;"
>
Pulisci
</button>


inviaDati(): void {
if (!this.utente.nome.trim() || !this.utente.cognome.trim()) {
alert('Compila tutti i campi prima di inviare!');
return;
}

console.log('Dati pronti nel TS grazie a ngModel:', this.utente);
this.messaggioInviato = `Inviato: ${this.utente.nome} ${this.utente.cognome}`;
}

/**
* Per svuotare i campi, ci basta azzerare le proprietà dell'oggetto nel TS.
* La View si adeguerà da sola all'istante.
*/
pulisciCampi(): void {
this.utente.nome = '';
this.utente.cognome = '';
this.messaggioInviato = '';
}


this.messaggioInviato = `Inviato: ${this.utente.nome} ${this.utente.cognome}`;
